进阶版

```shell
#!/bin/sh

# ============ Basic Configuration ============
NFS_SERVER="192.168.143.186"
NFS_PATH="/home/huangmingle"
MOUNT_POINT="/nfsroot"
CHIPNAME="xm7206v11a"
OSMEM="64M"
XMEDIA_KO_PATH="/komod"
#XMEDIA_KO_PATH="/nfsroot/projects/goke/7606v1/XMIPCLinuxV100R003C00SPC020/out/$CHIPNAME/ko"

ifconfig eth0 192.168.145.122 netmask 255.255.255.0 up
route add default gw 192.168.145.254
telnetd &

# ============ Layer 1: Wait for eth0 Link Up ============
echo "[net] Waiting for eth0 link up..."
TIMEOUT=3
while [ $TIMEOUT -gt 0 ]; do
    # Prefer reading sysfs (most reliable)
    if [ -f /sys/class/net/eth0/operstate ]; then
        STATE=$(cat /sys/class/net/eth0/operstate)
        [ "$STATE" = "up" ] && break
    else
        # Fallback: check the RUNNING flag in ifconfig output
        ifconfig eth0 | grep -q "RUNNING" && break
    fi
    sleep 1
    TIMEOUT=$((TIMEOUT - 1))
done

if [ $TIMEOUT -eq 0 ]; then
    echo "[net] WARN: eth0 link did not come up, continuing anyway..."
fi
echo "[net] eth0 link is up"

# ============ Layer 2: Wait for NFS server to be reachable ============
echo "[net] Waiting for NFS server ($NFS_SERVER) to be reachable..."
TIMEOUT=3
while [ $TIMEOUT -gt 0 ]; do
    if ping -c 1 -W 2 "$NFS_SERVER" >/dev/null 2>&1; then
        echo "[net] NFS server is reachable"
        break
    fi
    sleep 1
    TIMEOUT=$((TIMEOUT - 1))
done

if [ $TIMEOUT -eq 0 ]; then
    echo "[net] ERROR: NFS server unreachable, mount will likely fail"
fi

# ============ Layer 3: Mount NFS (with retries) ============
echo "[nfs] Mounting $NFS_SERVER:$NFS_PATH -> $MOUNT_POINT"
RETRIES=3
while [ $RETRIES -gt 0 ]; do
    if mount -t nfs -o nolock,tcp,addr=$NFS_SERVER \
         "$NFS_SERVER:$NFS_PATH" "$MOUNT_POINT" 2>/dev/null; then
        echo "[nfs] Mounted successfully"
        break
    fi
    echo "[nfs] Mount failed, retrying... ($RETRIES left)"
    sleep 3
    RETRIES=$((RETRIES - 1))
done

if [ $RETRIES -eq 0 ]; then
    echo "[nfs] FATAL: Failed to mount NFS after all retries"
fi

# ============ Load xmedia ko ============
cd $XMEDIA_KO_PATH
chmod +x load
./load $CHIPNAME -i -osmem $OSMEM
```
